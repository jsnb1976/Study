## Standard Module Call Pattern (for all 16 files)
```openscad
use <../lib/charlemagne_params.scad>
use <../lib/components/hull.scad>
use <../lib/components/alon_dome.scad>
use <../lib/components/aaj.scad>
use <../lib/components/cmbc_vault.scad>
use <../lib/components/hub_motor.scad>
use <../lib/components/leg.scad>
use <../lib/components/sensor_array.scad>
use <../lib/components/belly_pan.scad>

SF   = <value>;          // vessel scale factor
ef   = <0 | 0.4 | 1.0>; // explode factor for this view
gc   = p(GND_CLR, SF);   // ground clearance
hh   = p(HULL_H, SF);    // hull height
hw   = p(HULL_W, SF);    // hull width
hl   = p(HULL_L, SF);    // hull length
legs = <LEG_ANG_6|8|10>; // leg angle array

// Hull
translate([0, 0, gc + hh/2])
  charlemagne_hull(sf=SF);

// ALON dome
translate([0, 0, gc + hh + ef*EX_DOME*SF])
  alon_dome(sf=SF);

// AAJ joints (2 or 3)
for(x_sign = [1, -1])  // or [1,0,-1] for RHODONITE
  translate([x_sign * hl*0.28 + x_sign*ef*EX_AAJ*SF, 0, gc + hh*0.4])
    charlemagne_aaj(sf=SF, expanded=ef>0.5);

// CMBC vault(s)
translate([0, 0, gc - ef*EX_VAULT*SF])
  cmbc_vault(sf=SF);
// (RHODONITE: 4 vaults at ±offsets)

// Legs + motors
for(ang = legs) {
  rotate([0,0,ang])
    translate([p(LEG_REACH,SF) + ef*EX_LEG*SF, 0, gc])
      charlemagne_leg(sf=SF);
  rotate([0,0,ang])
    translate([p(LEG_REACH,SF) + ef*(EX_LEG+EX_MOTOR)*SF, 0, 0])
      hub_motor_assembly(sf=SF);
}

// Sensor array
translate([hl/2 + ef*EX_NOSE*SF, 0, gc + hh*0.3])
  sensor_array(sf=SF);

// Belly pan
translate([0, 0, gc - p(BELLY_T,SF)/2 - ef*EX_PON*SF])
  belly_pan(sf=SF);
```
